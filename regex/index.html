Daftar kecocokan

            
    
          
    
        
    
      
    

      
    
        
    Contekan
    
        
    Karakter
    
        
    
          
    .
    karakter apa saja, kecuali baris baru
    
          
    \d
    satu angka
    
          
    \w
    huruf, angka, atau garis bawah
    
          
    \s
    spasi, tab, atau baris baru
    
          
    [abc]
    salah satu dari a, b, c
    
          
    [^abc]
    karakter apa saja selain a, b, c
    
        
    
        
    Kuantifier
    
        
    
          
    *
    nol kali atau lebih
    
          
    +
    satu kali atau lebih
    
          
    ?
    nol atau satu kali
    
          
    {n,m}
    n sampai m kali
    
          
    *?
    seminimal mungkin (lazy)
    
        
    
        
    Jangkar
    
        
    
          
    ^
    awal teks, atau awal baris dengan flag m
    
          
    $
    akhir teks, atau akhir baris dengan flag m
    
          
    \b
    batas kata
    
        
    
        
    Grup dan lookahead
    
        
    
          
    (x)
    grup tangkap
    
          
    (?<nama>x)
    grup bernama
    
          
    (?:x)
    grup tanpa tangkap
    
          
    a|b
    a atau b
    
          
    x(?=y)
    x hanya bila diikuti y
    
          
    x(?!y)
    x hanya bila tidak diikuti y
    
          
    (?<=y)x
    x hanya bila didahului y
